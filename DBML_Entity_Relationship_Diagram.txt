Table hospitals {
  hopital_id integer [primary key]
  name varchar
  address varchar
  size integer
  type varchar
  acceditation_status varchar
}
Table people {
  person_id integer [primary key]
  name varchar
  date_of_birth date
  address varchar
  role varchar
  registered_id integer
}
Table prescription {
  prescription_id integer [primary key]
  patient_id integer
  doctor_id integer
  medication varchar
  prescription_date date
}
