Table hospitals {
  hospital_id integer [primary key]
  hospital_name varchar
  address varchar
  bed_count integer
  type varchar
  accreditation_status varchar
}
Table people {
  person_id integer [primary key]
  name varchar
  date_of_birth date
  address varchar
  role varchar
  doctor_id integer
  hospital_id integer
}
Table prescriptions {
  prescription_id integer [primary key]
  patient_id integer
  doctor_id integer
  medication varchar
  prescription_date date
}
